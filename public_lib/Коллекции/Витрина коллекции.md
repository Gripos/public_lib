---
title: Витрина коллекции
---

```base
filters:
  and:
    - file.path.contains("Коллекции/")
    - "!type.isEmpty()"
    - model_brand != "Welly"
views:
  - type: grouped-cards
    name: Витрина
    groupBy:
      property: model_brand
      direction: DESC
    order:
      - note.car_brand
      - note.sku
    sort:
      - property: note.model_brand
        direction: ASC
      - property: note.car_brand
        direction: ASC
      - property: note.model
        direction: ASC
    image: note.cover
    imageAspectRatio: 0.6
    cardSize: 340
  - type: table-desc
    name: Каталог
    groupBy:
      property: model_brand
      direction: DESC
    order:
      - file.name
      - car_brand
      - country
      - color
      - series
      - sku
      - status
      - purchaseDate
      - price
      - condition
      - scale
    sort:
      - property: scale
        direction: ASC
      - property: car_brand
        direction: ASC
      - property: model
        direction: ASC
      - property: purchaseDate
        direction: DESC
```
